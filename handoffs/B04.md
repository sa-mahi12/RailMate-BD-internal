## Negative tests (proposed — for coordinator to place under `test/`)
```dart
// Fake sender helpers; client constructed with anonKey: 'test-anon-key'
// (placeholder, never a real secret) and supabaseUrl: 'https://x.supabase.co'.
import 'package:flutter_test/flutter_test.dart';
import 'package:railmate_bd/features/graphql/graphql_client.dart';
import 'package:railmate_bd/features/graphql/station_graphql_service.dart';
import 'package:railmate_bd/features/search/models/station.dart';

Map<String, dynamic> okBody() => <String, dynamic>{
      'data': <String, dynamic>{
        'stationsCollection': <String, dynamic>{
          'edges': <dynamic>[
            <String, dynamic>{
              'node': <String, dynamic>{
                'id': '22222222-2222-2222-2222-222222222222',
                'code': 'DHK',
                'name': 'Dhaka'
              }
            },
            <String, dynamic>{
              'node': <String, dynamic>{
                'id': '11111111-1111-1111-1111-111111111111',
                'code': 'CTG',
                'name': 'Chattogram'
              }
            },
          ]
        }
      }
    };

StationGraphqlClient fakeClient(GraphqlSender sender) =>
    StationGraphqlClient(
        supabaseUrl: 'https://x.supabase.co',
        anonKey: 'test-anon-key',
        sender: sender);

void main() {
  test('parses stations and sorts by name', () async {
    final client = fakeClient((uri, headers, body) async => okBody());
    // also assert endpoint + auth headers + query string here:
    // expect(uri.toString(), 'https://x.supabase.co/graphql/v1');
    // expect(headers['apikey'], 'test-anon-key');
    // expect((body['query'] as String), contains('stationsCollection'));
    final stations = await client.queryStations();
    expect(stations.map((s) => s.code), <String>['CTG', 'DHK']);
  });

  test('NEGATIVE: GraphQL errors array -> GraphqlResponseError', () async {
    final client = fakeClient((uri, headers, body) async =>
        <String, dynamic>{
          'errors': <dynamic>[
            <String, dynamic>{'message': 'permission denied for table stations'}
          ]
        });
    expect(client.queryStations(), throwsA(isA<GraphqlResponseError>()));
  });

  test('NEGATIVE: missing data shape -> GraphqlMalformedError', () async {
    final client =
        fakeClient((uri, headers, body) async => <String, dynamic>{'data': null});
    expect(client.queryStations(), throwsA(isA<GraphqlMalformedError>()));
  });

  test('NEGATIVE: transport failure -> GraphqlNetworkError, REST fallback',
      () async {
    final client = fakeClient(
        (uri, headers, body) async =>
            throw const GraphqlNetworkError('boom', isTimeout: true),
        );
    final stations = await fetchStationListWithFallback(
      graphql: client,
      restFallback: () async => okBody()['data']['stationsCollection']['edges']
          .map((e) => Station.fromMap(Map<String, dynamic>.from(
              (e as Map)['node'] as Map)))
          .toList(),
    );
    expect(stations, hasLength(2));
  });

  test('NEGATIVE: response error does NOT fall back to REST', () async {
    final client = fakeClient((uri, headers, body) async =>
        <String, dynamic>{'errors': <dynamic>[{'message': 'bad field'}]});
    var restCalled = false;
    await expectLater(
      fetchStationListWithFallback(
        graphql: client,
        restFallback: () async {
          restCalled = true;
          return <Station>[];
        },
      ),
      throwsA(isA<GraphqlResponseError>()),
    );
    expect(restCalled, isFalse);
  });
}
```
